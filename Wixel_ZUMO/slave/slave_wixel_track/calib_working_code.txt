/*
 * Simple integer square root (for distance calculation if needed)
  */
int16 isqrt(int32 n)
{
    int32 x = n;
    int32 y = (x + 1) >> 1;
    while (y < x)
    {
        x = y;
        y = (x + n / x) >> 1;
    }
    return (int16)x;
}

/*
 * Get absolute value of int16
 */
int16 abs16(int16 val)
{
    return (val < 0) ? -val : val;
}

/*
 * Check if position is within threshold using simple delta comparison
 * Returns 1 if within threshold, 0 otherwise
 */
uint8 isWithinThreshold(int16 currentX, int16 currentY, int16 targetX, int16 targetY, int16 threshold)
{
    int16 dx = abs16(targetX - currentX);
    int16 dy = abs16(targetY - currentY);
    
    // Goal reached if both dx and dy are within threshold
    return (dx < threshold && dy < threshold);
}

/*
 * Filter position data (threshold-based jump detection)
 */
void filterPosition()
{
    int16 dx, dy;
    int32 distSquared;
    int16 alpha_num, alpha_den;
    
    dx = posX - filteredX;
    dy = posY - filteredY;
    distSquared = (int32)dx * dx + (int32)dy * dy;
    
    // If change is large (jump detected), use low alpha; otherwise high alpha
    if (distSquared > (int32)POSITION_JUMP_THRESHOLD * POSITION_JUMP_THRESHOLD)
    {
        // Use low alpha (0.2): filteredX = 0.2 * posX + 0.8 * filteredX
        alpha_num = 1;
        alpha_den = 5;
    }
    else
    {
        // Use high alpha (0.9): filteredX = 0.9 * posX + 0.1 * filteredX
        alpha_num = 9;
        alpha_den = 10;
    }
    
    filteredX = (int16)((alpha_num * posX + (alpha_den - alpha_num) * filteredX) / alpha_den);
    filteredY = (int16)((alpha_num * posY + (alpha_den - alpha_num) * filteredY) / alpha_den);
    filteredTheta = (int8)((alpha_num * posTheta + (alpha_den - alpha_num) * filteredTheta) / alpha_den);
}



/*
 * Initialize Timer3 for PWM output
 */
void timer3Init()
{
    T3CTL = 0b01110000;      // Prescaler 1:8, frequency = 11.7 kHz (matches alpha.c)
    T3CC0 = T3CC1 = 0;       // Set duty cycles to zero
    T3CCTL0 = T3CCTL1 = 0b00100100;    // Compare mode
    PERCFG &= ~(1<<5);       // Alternate location (matches alpha.c)
    P1SEL |= (1<<R_PWM_PIN) | (1<<L_PWM_PIN);  // P1_3 and P1_4 as PWM
}

 */

{
    radioRegistersInit();
    
    CHANNR = 128;
    PKTLEN = RADIO_PACKET_SIZE;     // HARDCODED: 64 bytes for 4 slaves
    
    MCSM0 = 0x14;
    MCSM1 = 0x00;
    
    dmaConfig.radio.DC6 = 19;
    dmaConfig.radio.SRCADDRH = XDATA_SFR_ADDRESS(RFD) >> 8;
    dmaConfig.radio.SRCADDRL = XDATA_SFR_ADDRESS(RFD);
    dmaConfig.radio.DESTADDRH = (unsigned int)rxPacket >> 8;
    dmaConfig.radio.DESTADDRL = (unsigned int)rxPacket;
    dmaConfig.radio.LENL = 1 + RADIO_PACKET_SIZE + 2;
    dmaConfig.radio.VLEN_LENH = 0b10000000;
    dmaConfig.radio.DC7 = 0x10;
    
    DMAARM |= (1<<DMA_CHANNEL_RADIO);
    RFST = 2;
}

/*
 * Convert APF velocity (apfVelX, apfVelY in mm/s) to motor commands
 * 
 * Robot kinematics (differential drive):
 * - Left wheel velocity = (forward velocity - angular velocity * wheelbase/2)
 * - Right wheel velocity = (forward velocity + angular velocity * wheelbase/2)
 * 
 * - velocity_forward = magnitude of (apfVelX, apfVelY)
 * - heading_desired = atan2(apfVelY, apfVelX)
 * - heading_current = posTheta
 * - steering = (heading_desired - heading_current)
 */
void apf2motor()
{
    uint32 velMagSquared;
    int32 absX, absY;
    int16 velMag;
    int16 heading_desired;
    int16 steering_error;
    int16 vel_left, vel_right;
    
    
    // Calculate magnitude of desired velocity (forward velocity)
    velMagSquared = ((int32)apfVelX * apfVelX) + ((int32)apfVelY * apfVelY);
    
    if (velMagSquared > 2147483647) 
    {
        // We are at max range, just use the sqrt of INT32_MAX
        velMag = 46340; // isqrt(2147483647)
    }
    else
    {
        velMag = isqrt((int32)velMagSquared);
    }

    if (velMag == 0)
    {
        // Stop motors
        T3CC0 = 0;
        T3CC1 = 0;
        P1_5 = 0;  // Right direction
        P1_6 = 0;  // Left direction
        return;
    }
    
    // Calculate desired heading using approximation
    // heading_desired ≈ atan2(apfVelY, apfVelX) in degrees
    // For small angles: atan2(y, x) ≈ (y / x) * 57.3 (radians to degrees)
    // Simplified: heading_desired = (apfVelY / apfVelX) * K
    if (apfVelX == 0)
    {
        heading_desired = (apfVelY > 0) ? 90 : -90;
    }
    else
    {
        // Calculate angle using approximation
        // atan(y/x) ≈ (y/x) * (45 - abs(y/x) * (45 - 57.3))
        // Simplified for embedded: use lookup or quadrant-based approach
        
        // Simple quadrant approach:
        absX = (apfVelX > 0) ? apfVelX : -apfVelX;
        absY = (apfVelY > 0) ? apfVelY : -apfVelY;
        
        if (absX > absY)
        {
            // More horizontal than vertical
            // angle ≈ (y / x) * 45
            heading_desired = (int16)((apfVelY * 45) / apfVelX);
        }
        else
        {
            // More vertical than horizontal
            // angle ≈ 90 - (x / y) * 45
            if (apfVelY > 0)
                heading_desired = 90 - (int16)((apfVelX * 45) / apfVelY);
            else
                heading_desired = -90 - (int16)((apfVelX * 45) / apfVelY);
        }
    }
    
    // Calculate steering error (desired heading - current heading)
    steering_error = heading_desired - (int16)filteredTheta;
    
    // Normalize steering error to [-180, 180]
    if (steering_error > 180)
        steering_error -= 360;
    if (steering_error < -180)
        steering_error += 360;
    
    // Differential drive kinematics
    // vel_left = forward_velocity - (steering_error * wheelbase / 2)
    // vel_right = forward_velocity + (steering_error * wheelbase / 2)
    vel_left = velMag - ((steering_error * WHEELBASE_MM * STEERING_GAIN) / 200);
    vel_right = velMag + ((steering_error * WHEELBASE_MM * STEERING_GAIN) / 200);
    
    // Clamp velocities to valid range
    if (vel_left > ZUMO_MAX_VELOCITY_MMS)
        vel_left = ZUMO_MAX_VELOCITY_MMS;
    if (vel_left < -ZUMO_MAX_VELOCITY_MMS)
        vel_left = -ZUMO_MAX_VELOCITY_MMS;
    if (vel_right > ZUMO_MAX_VELOCITY_MMS)
        vel_right = ZUMO_MAX_VELOCITY_MMS;
    if (vel_right < -ZUMO_MAX_VELOCITY_MMS)
        vel_right = -ZUMO_MAX_VELOCITY_MMS;
    
    // Convert velocities to PWM
    pwm_left = velocityToPWM(vel_left);
    pwm_right = velocityToPWM(vel_right);
}

void setMotorsPWM(){
    // Set left motor (P1_4 = PWM (T3CC1), P1_6 = direction)
    if (pwm_left >= 0)
    {
        T3CC1 = pwm_left;
        P1_6 = 0;  // Forward
    }
    else
    {
        T3CC1 = -pwm_left;
        P1_6 = 1;  // Backward
    }
    
    // Set right motor (P1_3 = PWM (T3CC0), P1_5 = direction)
    if (pwm_right >= 0)
    {
        T3CC0 = pwm_right;
        P1_5 = 0;  // Forward
    }
    else
    {
        T3CC0 = -pwm_right;
        P1_5 = 1;  // Backward
    }
}

/*
 * Stop motors immediately
 */
void stopMotors()
{
    T3CC0 = 0;
    T3CC1 = 0;
    P1_5 = 0;
    P1_6 = 0;
}

/* ================== APF (ATTRACTION ONLY) ================== */

/*
 * Compute attractive force toward goal (simple potential field)
 * F = k * (Goal - Current)
 * 
 * Input: current position (mm), target position (mm)
 * Output: desired velocity (mm/s) stored in apfVelX, apfVelY
 */
void computeAttractiveForcesGoTo()
{
    int32 dx, dy;
    int32 distSquared;
    int16 dist;
    int16 velMag;
    
    // Vector from current to goal
    dx = (int32)targetX - (int32)filteredX;
    dy = (int32)targetY - (int32)filteredY;
    
    distSquared = dx*dx + dy*dy;
    
    // Check if goal reached using simple threshold check
    if (isWithinThreshold(filteredX, filteredY, targetX, targetY, APF_GOAL_THRESHOLD))
    {
        apfVelX = 0;
        apfVelY = 0;
        return;
    }
    
    // Calculate distance
    dist = isqrt(distSquared);
    
    if (dist > 0)
    {
        // Desired velocity magnitude (proportional to distance, capped at APF_MAX_VELOCITY)
        // prolly cast error here NOTE
        velMag = (dist > APF_MAX_VELOCITY) ? APF_MAX_VELOCITY : dist;

        // Desired velocity direction (normalized)
        apfVelX = (int16)((dx * (int32)velMag) / (int32)dist);
        apfVelY = (int16)((dy * (int32)velMag) / (int32)dist);
    }
    else
    {
        apfVelX = 0;
        apfVelY = 0;
    }
}

/*
 * Extract position data from packet
 */
void extractPositionData(uint8 offset)
{
    posX = (int16)((rxPacket[offset + 1] << 8) | rxPacket[offset + 2]);
    posY = (int16)((rxPacket[offset + 3] << 8) | rxPacket[offset + 4]);
    posTheta = (int8)rxPacket[offset + 5];
}

/*
 * Process CMD_STOP command
 */
void handleCmdStop()
{
    currentState = STATE_IDLE;
    ledPurple = 0;
    waypointCount = 0;
    currentWaypointIndex = 0;
    apfVelX = 0;
    apfVelY = 0;
    pwm_left = 0;
    pwm_right = 0;
    stopMotors();  // Immediately stop motors
}

/*
 * Process CMD_GO_TO command
 * Extract target position from data bytes
 */
void handleCmdGoTo(uint8 offset)
{
    if (currentState == STATE_RUN)
    {
        // Ignore GO_TO if in RUN state
        return;
    }

    // Target X: Data[0:1] (bytes at offset+7:8)
    // Target Y: Data[2:3] (bytes at offset+9:10)
    targetX = (int16)((rxPacket[offset + 7] << 8) | rxPacket[offset + 8]);
    targetY = (int16)((rxPacket[offset + 9] << 8) | rxPacket[offset + 10]);
    

    // Only ignore GO_TO if we are already in HOME and the target hasn't changed.
    if (currentState == STATE_HOME && targetX == lastTargetX && targetY == lastTargetY)
    {
        return;  // Already navigating to this target; no change
    }
    
    if (currentState == STATE_HOME)
    {
        // Restart GO_TO only if target changes
        if (targetX != lastTargetX || targetY != lastTargetY)
        {
            stateStartTime = (uint32)getMs();
            lastTargetX = targetX;
            lastTargetY = targetY;
        }
    }
    else
    {
        // Entering HOME state from IDLE
        stateStartTime = (uint32)getMs();
        lastTargetX = targetX;
        lastTargetY = targetY;
    }

    currentState = STATE_HOME;
    ledPurple = 0;
    waypointCount = 0;
    waypointInputIndex = 0;
    currentWaypointIndex = 0;
    apfVelX = 0;
    apfVelY = 0;
}

/*
 * Process CMD_PREP command
 */
void handleCmdPrep(uint8 offset)
{
    if (currentState != STATE_IDLE)
    {
        // Only accept PREP from IDLE state
        return;
    }

    if (!ledPurple)
    {
        waypointCount = 0;
        ledPurple = 1;
    }
    
    // each packet can contain up to 2 waypoints
    // run is fully updated when waypointCount >= MAX_WAYPOINTS
    for (i = 0; i < 2; i++)
    {
        uint8 dataOffset = offset + 7 + (i * 5);
        waypointInputIndex = rxPacket[dataOffset + 0];
        
        waypoints[waypointInputIndex].x = (int16)((rxPacket[dataOffset + 1] << 8) | rxPacket[dataOffset + 2]);
        waypoints[waypointInputIndex].y = (int16)((rxPacket[dataOffset + 3] << 8) | rxPacket[dataOffset + 4]);
        
        waypointCount++;
    }
}

/*
 * Process CMD_RUN command
 */
void handleCmdRun()
{
    if (currentState != STATE_IDLE)
    {
        // Only accept RUN from IDLE state
        return;
    }
    // if RUN command received, but is currently in RUN state, then ignore
    if (waypointCount >= MAX_WAYPOINTS)
    {
        currentState = STATE_RUN;
        ledPurple = 0;
        currentWaypointIndex = 0;
        stateStartTime = (uint32)getMs();
        lastApfUpdateTime = (uint32)getMs();
    }
}

/*
 * Process CMD_AUX command (for auxiliary functions)
 */
void handleCmdAux(uint8 offset)
{
    uint8 auxSubCmd = rxPacket[offset + 7];
    
    switch(auxSubCmd)
    {
        case 0x9F: // Default update position from background
            break;

        case 0x0F:  // Position offset
            offsetX = (int16)((rxPacket[offset + 8] << 8) | rxPacket[offset + 9]);
            offsetY = (int16)((rxPacket[offset + 10] << 8) | rxPacket[offset + 11]);
            // LED confirmation (optional): flash GREEN 2x
            break;
            
        case 0x1F:  // Manual PWM - BLOCKING
            currentState = STATE_IDLE; // Ensure we are in IDLE
            manualPwmLeft = (int16)((rxPacket[offset + 8] << 8) | rxPacket[offset + 9]);
            manualPwmRight = (int16)((rxPacket[offset + 10] << 8) | rxPacket[offset + 11]);
            
            // BLOCKING execution
            executeManualPwm();
            break;
            
        case 0x2F:  // Future: calibration data from analysis
            // Placeholder - implement when you know what data you need
            break;
            
        default:
            // Unknown AUX command - ignore
            break;
    }
}


/*
 * Main packet reception handler
 */
void receiveAndProcessPackets()
{
    uint8 slaveAddress;
    uint8 cmd;
    uint8 packetOffset;
    if (RFIF & (1<<4))
    {
        if (radioCrcPassed())
        {
            lastPacketTime = (uint16)getMs();
            // Update RED LED pulse
            if (lastPacketTime != lastPacketTimeCheck)
            {
                rxPulseStart = 1; // reset the counter
                lastPacketTimeCheck = lastPacketTime;
            }

            // Determine slave offset
            if (THIS_SLAVE_ADDRESS == 0x01)
                packetOffset = 1;
            else if (THIS_SLAVE_ADDRESS == 0x02)
                packetOffset = 17;
            else if (THIS_SLAVE_ADDRESS == 0x03)
                packetOffset = 33;
            else if (THIS_SLAVE_ADDRESS == 0x04)
                packetOffset = 49;
            else
            {
                // NOTE we do not handle more than 4 slaves
                RFIF &= ~(1<<4);
                DMAARM |= (1<<DMA_CHANNEL_RADIO);
                RFST = 2;
                return;
            }

            
            slaveAddress = rxPacket[packetOffset + 0];
            cmd = rxPacket[packetOffset + 6];
            
            if (slaveAddress != THIS_SLAVE_ADDRESS)
            {
                // Whole message is maybe wrong
                RFIF &= ~(1<<4);
                DMAARM |= (1<<DMA_CHANNEL_RADIO);
                RFST = 2;
                return;
            }
            
            // Extract and filter position for this address
            extractPositionData(packetOffset);
            filterPosition();

            // TODO, also extract and update the position of the other slaves

            
            // Process command
            switch(cmd)
            {
                case CMD_STOP:
                    handleCmdStop();
                    break;
                case CMD_GO_TO:
                    handleCmdGoTo(packetOffset);
                    break;
                case CMD_PREP:
                    handleCmdPrep(packetOffset);
                    break;
                case CMD_RUN:
                    handleCmdRun();
                    break;
                case CMD_AUX:
                    handleCmdAux(packetOffset);
                    break;
            }
        }
        
        RFIF &= ~(1<<4);
        DMAARM |= (1<<DMA_CHANNEL_RADIO);
        RFST = 2;
    }
}

/*
 * Handle connection timeout
 */
void handlePacketTimeout()
{
    if ((uint16)(getMs() - lastPacketTime) > CONNECTION_TIMEOUT)
    {
        currentState = STATE_IDLE;
        ledPurple = 0;
        waypointCount = 0;
        apfVelX = 0;
        apfVelY = 0;
        pwm_left = 0;
        pwm_right = 0;
        stopMotors();  // Immediately stop motors
    }
}

 * Update state machine
 */
void updateStateMachine()
{
    switch(currentState)
    {
